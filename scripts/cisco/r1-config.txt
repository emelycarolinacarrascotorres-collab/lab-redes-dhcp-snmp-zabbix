 description WAN-INTERNET

 no shutdown
!
! --- LAN del laboratorio (direccionamiento derivado de la matricula 2025-0697)
interface GigabitEthernet1/0
 description LAN-2025-0697
 ip address 20.25.6.97 255.255.255.0

 no shutdown
!

ip dhcp excluded-address 20.25.6.1 20.25.6.99
ip dhcp pool LAN-0697
 network 20.25.6.0 255.255.255.0

 lease 7
!
! --- NAT de la LAN hacia la WAN
access-list 1 permit 20.25.6.0 0.0.0.255
ip nat inside source list 1 interface FastEthernet0/0 overload
!
! --- SNMP de solo lectura (valores de laboratorio)
